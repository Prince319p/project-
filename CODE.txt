        const generateEmilyDataset = () => {
            const data = [];
            const startDate = new Date('2025-01-01');
            const exerciseTypes = ['Weight Training', 'Dancing', 'Walking', 'Running', 'Yoga', 'None'];
            for (let i = 0; i < 365; i++) {
                const currentDate = new Date(startDate);
                currentDate.setDate(startDate.getDate() + i);
                const dateStr = currentDate.toISOString().split('T')[0];
                const steps = Math.floor(4000 + Math.random() * 8000);
                const distance_km = parseFloat((steps * 0.00075).toFixed(2));
                const calories_burned = Math.floor(1800 + steps * 0.05 + Math.random() * 300);
                const exercise_minutes = Math.floor(20 + Math.random() * 60);
                const sleep_hours = parseFloat((5.5 + Math.random() * 3.5).toFixed(1));
                const sleep_score = Math.min(100, Math.floor((sleep_hours / 8) * 70 + 25));
                data.push({
                    date: dateStr,
                    steps,
                    distance_km,
                    calories_burned,
                    exercise_minutes,
                    exercise_type: exerciseTypes[Math.floor(Math.random() * exerciseTypes.length)],
                    avg_heart_rate_bpm: Math.floor(70 + Math.random() * 15),
                    resting_heart_rate_bpm: Math.floor(58 + Math.random() * 8),
                    hrv_ms: Math.floor(45 + Math.random() * 40),
                    sleep_hours,
                    deep_sleep_pct: parseFloat((14 + Math.random() * 14).toFixed(1)),
                    sleep_score,
                    stress_score: Math.floor(2 + Math.random() * 7),
                    water_liters: parseFloat((1.5 + Math.random() * 1.8).toFixed(1)),
                    weight_kg: parseFloat((61.5 + Math.sin(i / 30) * 1.2).toFixed(1)),
                    mood_1to5: Math.floor(2 + Math.random() * 3)
                });
            }
            return data;
        };

        function initNavigation() {
            document.querySelectorAll('.nav-item').forEach(item => {
                item.addEventListener('click', (e) => {
                    e.preventDefault();
                    const targetTab = item.getAttribute('data-tab');
                    document.querySelectorAll('.nav-item').forEach(n => n.classList.remove('active'));
                    item.classList.add('active');
                    document.querySelectorAll('.tab-content').forEach(t => t.classList.add('hidden'));
                    const activeTab = document.getElementById(`tab-${targetTab}`);
                    if (activeTab) activeTab.classList.remove('hidden');
                    state.activeTab = targetTab;
                    renderDashboard();
                });
            });

            });

        function initDateFilters() {
            const startInput = document.getElementById('date-start');
            const endInput = document.getElementById('date-end');
            const applyBtn = document.getElementById('apply-custom-date');

            if (applyBtn && startInput && endInput) {
                applyBtn.addEventListener('click', () => {
                    filterDataset(startInput.value, endInput.value);
                });
            }

            document.querySelectorAll('.filter-btn').forEach(btn => {
                btn.addEventListener('click', () => {
                    document.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active', 'bg-white', 'text-indigo-600', 'shadow-sm'));
                    btn.classList.add('active', 'bg-white', 'text-indigo-600', 'shadow-sm');
                    const days = parseInt(btn.getAttribute('data-filter'));
                    const end = new Date('2025-12-31');
                    const start = new Date('2025-12-31');
                    start.setDate(end.getDate() - days + 1);

                    const startStr = start.toISOString().split('T')[0];
                    const endStr = end.toISOString().split('T')[0];

                    if (startInput) startInput.value = startStr;
                    if (endInput) endInput.value = endStr;

                    filterDataset(startStr, endStr);
                });
            });
        }

        function filterDataset(startDateStr, endDateStr) {

                state.filteredDataset = state.rawDataset.slice(-7);
            }
            renderDashboard();
        }

        function computeSummaryMetrics(dataset) {
            const count = dataset.length || 1;
            return {
                avgSteps: Math.round(dataset.reduce((a, b) => a + b.steps, 0) / count),
                totalDistance: dataset.reduce((a, b) => a + b.distance_km, 0).toFixed(1),
                avgCalories: Math.round(dataset.reduce((a, b) => a + b.calories_burned, 0) / count),
                totalCalories: dataset.reduce((a, b) => a + b.calories_burned, 0),
                avgExerciseMins: Math.round(dataset.reduce((a, b) => a + b.exercise_minutes, 0) / count),
                totalExerciseMins: dataset.reduce((a, b) => a + b.exercise_minutes, 0),
                avgSleepHours: (dataset.reduce((a, b) => a + b.sleep_hours, 0) / count).toFixed(1),
                avgSleepScore: Math.round(dataset.reduce((a, b) => a + b.sleep_score, 0) / count),
                avgDeepSleepPct: (dataset.reduce((a, b) => a + b.deep_sleep_pct, 0) / count).toFixed(1),
                avgStress: (dataset.reduce((a, b) => a + b.stress_score, 0) / count).toFixed(1),
                avgHRV: Math.round(dataset.reduce((a, b) => a + b.hrv_ms, 0) / count),
                avgRestingHR: Math.round(dataset.reduce((a, b) => a + b.resting_heart_rate_bpm, 0) / count),
                avgWater: (dataset.reduce((a, b) => a + b.water_liters, 0) / count).toFixed(1),
                latestWeight: dataset[dataset.length - 1]?.weight_kg || 62.0
            };
        }

        function renderDashboard() {
            const summary = computeSummaryMetrics(state.filteredDataset);
            if (state.activeTab === 'overview') {
                renderOverviewKPIs(summary);
                renderOverviewCharts(summary);
            } else if (state.activeTab === 'analytics') {
                renderAnalyticsView(summary);
            } else if (state.activeTab === 'sleep') {
                renderSleepView(summary);
            } else if (state.activeTab === 'activity') {

        function renderSmartAlerts() {
            const list = document.getElementById('smart-alerts-list');
            if (!list) return;

            const summary = computeSummaryMetrics(state.filteredDataset);
            const alerts = [];

            if (parseFloat(summary.avgStress) > 5.5) {
                alerts.push({ title: 'Elevated Stress Alert', msg: `Average stress level is currently ${summary.avgStress}/10 across the selected date range.` });
            }
            if (parseFloat(summary.avgSleepHours) < 7.0) {
                alerts.push({ title: 'Sleep Deficit Notice', msg: `Average sleep duration (${summary.avgSleepHours} hrs) is below your target.` });
            }
            if (summary.avgSteps >= state.goals.steps) {
                alerts.push({ title: 'Step Goal Achieved', msg: `Great job! Your average steps (${summary.avgSteps.toLocaleString()}) hit your daily target!` });
            }

            if (alerts.length === 0) {
                list.innerHTML = `<div class="p-4 rounded-xl bg-slate-100 dark:bg-slate-800 text-slate-600 dark:text-slate-300 text-xs font-semibold">All metrics are currently within healthy target ranges.</div>`;
            } else {
                list.innerHTML = alerts.map(a => `
                    <div class="p-4 rounded-xl bg-slate-100 dark:bg-slate-800/80 border border-slate-200 dark:border-slate-700 text-xs text-slate-800 dark:text-slate-200">
                        <h4 class="font-bold text-indigo-600 dark:text-indigo-400 mb-1">${a.title}</h4>
                        <p>${a.msg}</p>
                    </div>
                `).join('');
            }
        }

            const form = document.getElementById('ai-chat-form');

                                    <i class="fa-solid fa-volume-high"></i> Listen
                                </button>
                            </div>
                        </div>
                    `;
                }
                chatContainer.appendChild(msgDiv);
                chatContainer.scrollTop = chatContainer.scrollHeight;
                return msgDiv;
            };

            form.addEventListener('submit', async (e) => {
                e.preventDefault();
                const userPrompt = input.value.trim();
                if (!userPrompt) return;

                appendMessage('user', userPrompt);
                input.value = '';

                const summary = computeSummaryMetrics(state.filteredDataset);

                const loadingMsg = appendMessage('ai', '<span class="italic text-slate-400"><i class="fa-solid fa-spinner fa-spin mr-1"></i> Gemini AI is analyzing your dataset & request...</span>');

                try {
                    const responseText = await getGeminiOrLocalResponse(userPrompt, summary);

                    appendMessage('ai', responseText);
                } catch (err) {
                    loadingMsg.remove();
                    appendMessage('ai', generateComprehensiveHealthAnswer(userPrompt, summary));
                }
            });

            document.querySelectorAll('.ai-prompt-btn').forEach(btn => {
                btn.addEventListener('click', () => {

                });
            });

                systemInstruction: { parts: [{ text: systemPrompt }] }
            };

                const res = await fetch(apiUrl, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });

                if (res.ok) {
                    const data = await res.json();
                    const candidate = data.candidates?.[0];
                    if (candidate && candidate.content?.parts?.[0]?.text) {

                        text = text
                            .replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>')

                            .replace(/\n/g, '<br>');
                        return text;
                    }
                }
